cmake_minimum_required(VERSION 3.20)
project(tabby_freerdp LANGUAGES C)
find_package(OpenSSL REQUIRED)
if(FREERDP_SOURCE_DIR)
    include(${CMAKE_CURRENT_SOURCE_DIR}/options.cmake)
    add_subdirectory(${FREERDP_SOURCE_DIR} freerdp)
    set(TRD_LIBRARIES freerdp-client freerdp winpr)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FREERDP REQUIRED IMPORTED_TARGET freerdp3>=3.17 freerdp-client3 winpr3)
    set(TRD_LIBRARIES PkgConfig::FREERDP)
endif()
add_executable(tabby-freerdp main.c)
if(FREERDP_SOURCE_DIR)
    target_include_directories(tabby-freerdp PRIVATE ${FREERDP_SOURCE_DIR}/include
        ${CMAKE_CURRENT_BINARY_DIR}/freerdp/include ${FREERDP_SOURCE_DIR}/winpr/include
        ${CMAKE_CURRENT_BINARY_DIR}/freerdp/winpr/include)
endif()
target_compile_features(tabby-freerdp PRIVATE c_std_11)
target_link_libraries(tabby-freerdp PRIVATE ${TRD_LIBRARIES} OpenSSL::SSL OpenSSL::Crypto)
if(MSVC)
    target_compile_options(tabby-freerdp PRIVATE /W4)
else()
    target_compile_options(tabby-freerdp PRIVATE -Wall -Wextra -Werror=implicit-function-declaration)
endif()
install(TARGETS tabby-freerdp RUNTIME DESTINATION .)
